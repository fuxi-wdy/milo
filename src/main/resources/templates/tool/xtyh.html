1.TCP连接相关参数

vim /etc/sysctl.conf

增加以下几行：

net.ipv4.tcp_fin_timeout=300
net.ipv4.tcp_keepalive_time=1200
net.ipv4.tcp_syncookies=1
net.ipv4.tcp_tw_reuse=1
net.ipv4.tcp_tw_recycle=1
net.ipv4.ip_local_port_range=32768 61000
net.ipv4.tcp_max_tw_buckets=5000
fs.epoll.max_user_instances=20480
net.ipv4.tcp_max_syn_backlog=20480
net.core.somaxconn=10240
vm.swappiness=40
vm.vfs_cache_pressure=200
net.ipv4.ip_forward=1

执行生效

/sbin/sysctl -e -p /etc/sysctl.conf

3.数据库优化

【调整/dev/shm大小】

mount -o remount,size=6G /dev/shm
/etc/fstab
tmpfs /dev/shm tmpfs defaults,size=6G 0 0

【修改SGA,PAG】10G

alter system set sga_target=内存*0.8M scope=spfile;
alter system set sga_max_size=内存*0.8M scope=spfile;

alter system set pga_aggregate_target=内存*0.2M scope=spfile

【修改连接数】

alter system set processes=1000 scope=spfile;
alter system set sessions=1105 scope=spfile;

重启数据库

shutdown immediate
